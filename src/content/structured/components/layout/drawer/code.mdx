## Variants

The drawer components in the following code examples have the `boundary` prop set to `"parent"` so that they are contained within each component preview. To anchor the drawer to the viewport instead, set the `boundary` prop to `"viewport"`.

export const ControlledTriggerDrawer = () => {
  const [drawerExpanded, setDrawerExpanded] = useState(false);
  const toggleDrawer = () => {
    setDrawerExpanded(!drawerExpanded);
  };
  const handleDrawerExpanded = (ev) => {
    setDrawerExpanded(ev.detail.expanded);
  };
  return (
    <div style={{ position: "relative", height: "100%", width: "100%" }}>
      <IcButton onClick={toggleDrawer} style={{ margin: "1rem" }}>
        Find out more about roasted coffee
      </IcButton>
      <IcDrawer
        heading="Roasted coffee"
        message="Contrary to popular belief, light roast coffee has more caffeine than dark roast coffee. The longer coffee is roasted, the more caffeine is lost through the cooking of the bean. Light roast coffee is a light brown colour and has no oil on the surface of the beans, and these coffees typically have a crisp acidity, a mellow body, and bright flavours."
        boundary="parent"
        trigger="controlled"
        expanded={drawerExpanded}
        onIcDrawerExpanded={handleDrawerExpanded}
        style={{ "--ic-z-index-drawer": 0 }}
      >
        <SlottedSVG
          path={mdiCoffee}
          slot="heading-adornment"
          viewBox="0 0 24 24"
          height="1.5rem"
          width="1.5rem"
        />
        <IcButton slot="actions">Add to order</IcButton>
      </IcDrawer>
    </div>
  );
};
